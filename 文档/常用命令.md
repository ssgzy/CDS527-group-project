# 常用命令

## 环境启动
```bash
# 激活 CDS527 环境
conda activate CDS527

# 启动 JupyterLab（在项目根目录下运行）
jupyter lab

# 在不激活环境的情况下单次运行
conda run -n CDS527 python3 script.py
conda run -n CDS527 jupyter lab
```

## 数据快速检查
```bash
# CSV 行数、列名、标签分布
conda run -n CDS527 python3 - <<'PY'
import pandas as pd
df = pd.read_csv('smile-annotations-final.csv')
print(f"Shape: {df.shape}")
print(f"Columns: {df.columns.tolist()}")
print(df['emotions'].value_counts())
PY

# 文本长度统计
conda run -n CDS527 python3 - <<'PY'
import pandas as pd
df = pd.read_csv('smile-annotations-final.csv')
lengths = df['text'].str.split().str.len()
print(f"min={lengths.min()}, max={lengths.max()}, mean={lengths.mean():.2f}, median={lengths.median()}")
PY
```
